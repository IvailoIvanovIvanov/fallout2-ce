# SETUP INSTRUCTIONS
# ============================================================================
#
# 1. Clone FSR2 repository:
#    cd third_party
#    git clone https://github.com/GPUOpen-Effects/FidelityFX-SuperResolution.git fsr2
#    cd fsr2 && git checkout tags/v2.1.0
#
# 2. FSR2 has its own CMakeLists.txt. When you add_subdirectory(third_party/fsr2),
#    CMake will automatically build the FFX libraries.
#
# 3. The main library target is typically named: ffx_fsr2
#
# 4. Include path: third_party/fsr2/sdk/include
#    - Contains: ffx_fsr2.h, ffx_interface.h, etc.
#
# 5. For platform-specific backend setup, also need:
#    - DirectX 12:   third_party/fsr2/sdk/include_d3d12
#    - Vulkan:       third_party/fsr2/sdk/include_vk
#    - Other APIs as available

# BACKEND SELECTION
# ============================================================================
#
# FSR2 requires a graphics API backend. Update src/upscaler.cc to select:
#
# Windows (DirectX 12):
#     #include "ffx_fsr2_d3d12.h"
#     FfxInterface backend = ffxGetInterfaceD3D12(...);
#
# Linux/Mac (Vulkan):
#     #include "ffx_fsr2_vk.h"
#     FfxInterface backend = ffxGetInterfaceVk(...);
#
# Get device from SDL renderer:
#     SDL_Renderer* renderer = gSdlRenderer;
#     SDL_GetWindowFlags(gSdlWindow);  // Check for Vulkan vs D3D12

# SAMPLE MINIMAL CMakeLists.txt INTEGRATION
# ============================================================================

if(FALLOUT_VENDORED AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/fsr2/CMakeLists.txt")
    # Build vendored FSR2
    add_subdirectory(third_party/fsr2 EXCLUDE_FROM_ALL)
    
    # Create an imported target alias if not already created by FSR CMakeLists
    if(NOT TARGET FSR2::FSR2)
        add_library(FSR2::FSR2 ALIAS ffx_fsr2)
    endif()
    
    # Include FSR2 headers
    target_include_directories(${EXECUTABLE_NAME} PRIVATE 
        ${CMAKE_CURRENT_SOURCE_DIR}/third_party/fsr2/sdk/include
    )
    
    # Link FSR2
    target_link_libraries(${EXECUTABLE_NAME} PRIVATE ffx_fsr2)
    
    # Compile definition for upscaler.cc
    target_compile_definitions(${EXECUTABLE_NAME} PRIVATE FALLOUT_VENDORED_FSR2=1)
    
else()
    # Try system-wide FSR2
    find_package(FSR2)
    if(FSR2_FOUND)
        target_include_directories(${EXECUTABLE_NAME} PRIVATE ${FSR2_INCLUDE_DIRS})
        target_link_libraries(${EXECUTABLE_NAME} PRIVATE ${FSR2_LIBRARIES})
    else()
        message(WARNING "FSR2 not found - upscaling will not be available")
    endif()
endif()
